WebSocket Connection for Notifications

WebSocket connection for real-time notifications. Replace the access token with a valid token.

WebSocket URL:

ws://{base_url}/ws/notifications/?token=${accessToken}

WebSocket Consumer (Django Channels):
    class NotificationConsumer(AsyncWebsocketConsumer):
        # ... (Code for handling WebSocket connections and notifications)
React Implementation:
    const accessToken = "access_token_here";
    const wsURL = `ws://{base_url}/ws/notifications/?token=${accessToken}`;
    const socket = new WebSocket(wsURL);
    
    socket.addEventListener('open', (event) => {
        console.log('WebSocket connected:', event);
    });
    
    socket.addEventListener('message', (event) => {
        const data = JSON.parse(event.data);
        console.log('Message received:', data);
        if (data.hasOwnProperty('message') && Array.isArray(data.message)) {
            const notifications = data.message;
            notifications.forEach(notification => displayNotification(notification));
        } else {
            console.error('Invalid data format:', notifications);
        }
    });
    
    socket.addEventListener('close', (event) => {
        console.log('WebSocket closed:', event);
    });
    
    function displayNotification(notification) {
        // Implementation to display notifications in your React app
    }
    
    function sendNotificationId(notificationId) {
        const data = {
            notification_id: notificationId
        };
        socket.send(JSON.stringify(data));
    }
